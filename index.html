            <h2 class="description-heading">Project Description</h2>

            <p class="description-text">

              In English, we rely on subject pronouns to know who's doing the action: "I speak" vs. "you speak." Spanish often drops that pronoun entirely because the verb ending itself signals who is doing the action. Sometimes, verb endings will differ in stress alone, like <em><strong>HA</strong>-blo</em> ("I speak") vs. <em>ha-<strong>BLÓ</strong></em> ("he/she spoke"). These contrasts can be difficult for English-speaking learners to detect, because stress-based contrasts are less frequent in English.

              <br><br>

              The goal of this study is to examine how successfully Spanish learners at various proficiency levels process spoken Spanish verb forms that differ only in lexical stress. Our goal is to determine the best instructional level to introduce stress perception training.

<script>

  const workspace = document.getElementById('workspace');

  const cards = Array.from(document.querySelectorAll('.project-card'));

  const stageItems = Array.from(document.querySelectorAll('.stage-item'));

  const nowViewing = document.getElementById('now-viewing');


  function updateSidebar(card) {

    const projectName = card.dataset.project;

    const activeStages = card.dataset.stages.split(',');


    nowViewing.textContent = projectName;


    stageItems.forEach((item, i) => {

      item.classList.toggle('active', activeStages.includes(String(i + 1)));

    });

  }


  const observer = new IntersectionObserver((entries) => {

    const visible = entries

      .filter(e => e.isIntersecting)

      .sort((a, b) => b.intersectionRatio - a.intersectionRatio);

    if (visible.length) updateSidebar(visible[0].target);

  }, { root: workspace, threshold: [0.3, 0.5, 0.7] });


  cards.forEach(card => observer.observe(card));


  // Init

  if (cards.length) updateSidebar(cards[0]);

</script>
